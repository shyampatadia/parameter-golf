\begin{frame}{The goal}
\centering
\vspace{0.3em}
\Large
\textbf{Train a useful language model that fits in 16~MB.}\\[1.0em]

\normalsize
\hil{1{,}000$\times$ smaller than GPT-2.}\\
Smaller than an MP3 song.\\[1.4em]

Why this is interesting:
\begin{itemize}
  \item Tests which design decisions actually matter when capacity is scarce.
  \item Every megabyte saved on one component has to be earned somewhere else.
  \item A working model at this size is small enough to ship inside an app
        or run on-device.
\end{itemize}

\vspace{0.8em}
\textbf{The reference to beat:} \hil{1.2244\bpb} (the published INT8 baseline).
\end{frame}

% --------------------------------------------------------------- 3
\begin{frame}{What does ``score'' mean? -- bits per byte}
\begin{columns}[T,onlytextwidth]
\column{0.50\textwidth}
\small
\textbf{bpb} = bits the model spends to \emph{guess each byte} of unseen text.
Same idea as a compression ratio.

\vspace{0.5em}
\textbf{How it's calculated:}
\begin{enumerate}\setlength{\itemsep}{1pt}
  \item Model assigns probability $p$ to the next token.
  \item Bits spent on that token $=$ \hil{$-\log_2 p$}.
  \item Average over held-out tokens, normalize per byte.
\end{enumerate}

\vspace{0.5em}
{\footnotesize \textbf{Anchors for $-\log_2 p$:} \\
\hspace*{0.5em}$p{=}1.0 \to $ \good{0 bits} (perfect guess) \\
\hspace*{0.5em}$p{=}0.5 \to $ 1 bit \\
\hspace*{0.5em}$p{=}0.01 \to $ \bad{6.6 bits} (badly wrong)}

\vspace{0.5em}
\textcolor{accent}{\textbf{Lower is better.}} Every 0.01\bpb is measurable.

\column{0.50\textwidth}
\centering
\textbf{\small Worked example}
\vspace{0.2em}

\begin{tikzpicture}[every node/.style={font=\scriptsize, align=left}]
\node[draw=primary, very thick, rounded corners, fill=soft,
      text width=5.5cm, inner sep=6pt] (ex) {%
\textbf{Prefix:} \texttt{"The quick brown\_\_\_"} \\[0.3em]
\textbf{Model's top predictions:}\\
\hspace*{0.5em}\texttt{fox}\hspace{0.5em}$p{=}0.62$ \\
\hspace*{0.5em}\texttt{dog}\hspace{0.5em}$p{=}0.08$ \\
\hspace*{0.5em}\texttt{cat}\hspace{0.5em}$p{=}0.05$ \\[0.3em]
\textbf{If the actual next token is \texttt{fox}:} \\
\hspace*{0.5em}bits spent $=-\log_2(0.62) \approx$ \good{0.69} \\[0.3em]
\textbf{If it were \texttt{cat}:} \\
\hspace*{0.5em}bits spent $=-\log_2(0.05) \approx$ \bad{4.32} \\[0.3em]
Average across thousands of held-out tokens $\rightarrow$ \hil{bpb}.};
\end{tikzpicture}

\vspace{0.4em}
{\scriptsize\textbf{Reference scale:} \,
\bad{8.0} random \,$\cdot$\,
\hil{1.22} baseline \,$\cdot$\,
\good{1.0} GPT-4 \,$\cdot$\,
0 oracle}
\end{columns}
\end{frame}

\begin{frame}{What is this model, and what does it do?}
\begin{columns}[T,onlytextwidth]
\column{0.52\textwidth}
\textbf{A decoder-only Transformer} -- same family as GPT.\\[0.4em]
Its single capability:
\begin{itemize}\setlength{\itemsep}{1pt}
  \item \hil{Read a sequence of tokens.}
  \item \hil{Output a probability distribution over the next token.}
\end{itemize}
\emph{Generation} = sample from the distribution, append the token, repeat
-- the same loop used by every modern LM.

\vspace{0.6em}
\textbf{Trained from scratch on FineWeb} \\
{\footnotesize (a 10B-token open-web English shard from HuggingFace).}\\
\textbf{Tokenizer:} 1{,}024-piece SentencePiece BPE.\\
\hil{No pretrained weights. No transfer learning.}

% --------------------------------------------------------------- 5
\begin{frame}{Question 1: How aggressively can we shrink the weights?}
{\small\textbf{Quantization} = store each weight in fewer bits.
At 16~MB, INT8 fits $\sim$8M params, INT6 fits $\sim$11M.
Lower precision \emph{should} be a free win.}

\vspace{0.4em}
\begin{columns}[T,onlytextwidth]
\column{0.5\textwidth}
\centering
\includegraphics[width=\linewidth]{curves_o1_quant.pdf}

\column{0.5\textwidth}
\small
In training: INT6 and INT8 reach the \emph{same} loss.\\[0.4em]

After round-trip through \texttt{zlib}:
\begin{itemize}\setlength{\itemsep}{1pt}
  \item INT8: \good{1.216\bpb}
  \item INT6: \bad{1.349\bpb}
\end{itemize}

\vspace{0.3em}
The 6-bit bins can't represent the trained weights, and QAT doesn't recover the gap.

\vspace{0.3em}
\hil{More bits to spend $\neq$ more capacity if you can't read them back.}
\end{columns}
\end{frame}

% --------------------------------------------------------------- 6
\begin{frame}{Question 2: Does the rest of the design matter?}
\textbf{Three more decisions, one at a time:}

\vspace{0.6em}
\begin{itemize}\setlength{\itemsep}{6pt}
  \item \textbf{Position encoding} -- how the model knows token order.\\
        {\small Tested: full RoPE, partial RoPE, learned absolute, none.}\\
        \good{Partial RoPE wins at 1.215\bpb} -- but the gap between any two
        \emph{position-aware} schemes is tiny.\\
        Lesson: \emph{having} positional info matters; the form is second-order.

  \item \textbf{Optimizer} -- AdamW vs.\ Muon vs.\ Adafactor.\\
        Muon (orthogonalized gradient steps) wins by \good{$>$0.2\bpb}. \\
        Lesson: \emph{the optimizer is load-bearing, not a free knob.}

  \item \textbf{Weight factorizations} -- low-rank, block-diagonal, random
        projection. Force structure to save parameters.\\
        All land at \bad{1.30--1.40\bpb}. \\
        Lesson: \emph{at this scale, structured weights don't compete with
        dense ones.}
\end{itemize}
\end{frame}

% --------------------------------------------------------------- 7
\begin{frame}{Question 3: Can we just stack the winners?}
\textbf{Natural next move:} combine everything that worked --\\
partial RoPE \emph{plus} INT6 (more bits) \emph{plus} 11 layers \emph{plus} a wider MLP.

\vspace{0.8em}
\centering
\begin{tikzpicture}[every node/.style={font=\small, align=center}]
\node[draw=ok, very thick, rounded corners, fill=ok!8,
      minimum width=4.0cm, minimum height=1.0cm] (a)
      {Capacity expansion \\ \good{$-$0.063\bpb training}};
\node[draw=accent, very thick, rounded corners, fill=accent!8,
      minimum width=4.0cm, minimum height=1.0cm, right=1.0cm of a] (b)
      {INT6 quantizer \\ \bad{$+$0.120\bpb roundtrip}};
\node[draw=ink, thick, rounded corners,
      minimum width=2.6cm, minimum height=1.0cm, right=1.0cm of b] (c)
      {\Large $=$ \;\bad{net loss}};

\end{tikzpicture}

\vspace{0.8em}
Training-time loss \good{1.1614\bpb} -- the best we'd ever seen.\\
Final loss \bad{1.2816\bpb} -- worse than the simple baseline.

\vspace{0.8em}
\hil{Which lever is responsible -- the capacity, or the quantizer?}
\end{frame}

% --------------------------------------------------------------- 8
\begin{frame}{The answer: change one thing at a time}
Two follow-ups, each isolating \emph{one} side of the trade.

\vspace{0.4em}
\begin{columns}[T,onlytextwidth]
\column{0.55\textwidth}
\centering
\includegraphics[width=\linewidth]{curves_o5_resolved.pdf}

\column{0.45\textwidth}
\textbf{(a) Same architecture, swap quantizer:}\\
{\footnotesize TurboQuant: rotate each weight row by a random matrix, then
quantize. Originally an \emph{inference-time KV-cache} method (Zandieh~2024);
applied here to weights at 4 bits.}\\
$\rightarrow$ \bad{3.60\bpb} final. The quantizer fails far below 8 bits.

\vspace{0.5em}
\normalsize
\textbf{(b) Same quantizer, ease architecture:}\\
INT8 $+$ 11~layers $+$ MLP$=$2.\\
$\rightarrow$ \good{1.194\bpb}, \emph{0.030 below the reference.}

\vspace{0.7em}
\hil{The quantizer was the binding constraint.}\\
{\footnotesize The architecture lever works -- as long as you don't crush
precision to fund it.}
\end{columns}
\end{frame}

% --------------------------------------------------------------- 9
\begin{frame}{What else we tried (gains, plus three honest nulls)}
\small
\begin{columns}[T,onlytextwidth]
\column{0.5\textwidth}
\textbf{Multi-Query Attention.}\\
{\scriptsize Share \emph{one} K/V projection across all attention heads.
Frees parameters that fund a 12-layer model.}\\
\good{1.198\bpb} at 19.56~MB.\\[0.2em]
\centering\includegraphics[width=0.95\linewidth]{curves_o6_mqa.pdf}

\column{0.48\textwidth}
\textbf{QK-gain initialization.}\\
{\scriptsize A learnable scalar on the query projection.}\\
Tried 1.5, 3.0, 5.0. Spread \alert{0.002\bpb}. \\
\hil{Null effect.}\\[0.5em]

\textbf{Learning-rate / warmup sweep.}\\
$3 \times 3$ grid. Spread \alert{0.0019\bpb}. \\
\hil{Null effect.} The default recipe is already optimal.\\[0.5em]

\textbf{Eval protocol matters.}\\
{\scriptsize Sliding-window evaluation (move 256 tokens at a time, every token sees more
context) scores \good{1.160\bpb} on the same weights -- but this is \emph{not}
leaderboard-comparable, since the official protocol uses non-overlapping windows.}
\end{columns}

\vspace{0.4em}
\hil{Three nulls reported as findings} -- they constrain the design space and save future work.
\end{frame}

% --------------------------------------------------------------- 12
\begin{frame}{What we learned}
\begin{enumerate}\setlength{\itemsep}{6pt}
  \item \hil{The quantizer is the binding constraint.}
        Below 8 bits, weight quantization breaks at this scale -- and no
        clever fix (QAT, random rotation) saves it.

  \item \hil{Architecture matters, but only if the quantizer can carry it.}
        The same 11-layer model fails under INT6 and wins under INT8.

  \item \hil{Inductive biases (RoPE, MQA, depth) matter at the margin.}
        Each is worth a hundredth or two of a bpb -- real but small.

  \item \hil{16~MB is a soft elbow on the frontier, not a hard cutoff.}
        Spending 3.3~MB more buys 0.020\bpb -- a continuous trade.

  \item \hil{Three nulls are findings too.}
        Weight factorizations, QK-gain init, LR/warmup sweeps. Saying ``no''
        rules out a region of the design space.
\end{enumerate}
\end{frame}
